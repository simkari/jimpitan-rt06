// Kode ini akan membuat menu khusus di Google Sheets Anda
function onOpen() {
  var ui = SpreadsheetApp.getUi();
  ui.createMenu('🛠️ Menu Jimpitan')
      .addItem('Tutup Buku & Rekap Bulan Ini', 'prosesTutupBuku')
      .addToUi();
}

function prosesTutupBuku() {
  var ui = SpreadsheetApp.getUi();
  
  // --- KODE PENGAMANAN PIN DIMULAI DI SINI ---
  var responsPin = ui.prompt("🔐 Keamanan Sistem", "Masukkan PIN Admin untuk melakukan Tutup Buku:", ui.ButtonSet.OK_CANCEL);
  
  // Jika tombol Cancel diklik
  if (responsPin.getSelectedButton() !== ui.Button.OK) return;
  
  var pinMasuk = responsPin.getResponseText();
  
  // GANTI "123456" DENGAN PIN RAHASIA ANDA
  if (pinMasuk !== "0607") { 
    ui.alert("Akses Ditolak", "PIN yang Anda masukkan salah. Proses Tutup Buku dibatalkan secara otomatis.", ui.ButtonSet.OK);
    return;
  }
  // --- KODE PENGAMANAN PIN SELESAI ---

  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheetSejarah = ss.getSheetByName("DATABASE_SEJARAH");
  // ... (lanjutkan dengan sisa kode asli di bawahnya) ...
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheetSejarah = ss.getSheetByName("DATABASE_SEJARAH");
  
  if (!sheetSejarah) {
    ui.alert("Silakan buat tab baru dengan nama 'DATABASE_SEJARAH' terlebih dahulu.");
    return;
  }
  
  // Meminta konfirmasi bulan dan tahun dari admin
  var responsBulan = ui.prompt("Tutup Buku", "Silakan masukkan nama Bulan (contoh: November):", ui.ButtonSet.OK_CANCEL);
  if (responsBulan.getSelectedButton() !== ui.Button.OK) return;
  var bulan = responsBulan.getResponseText();
  
  var responsTahun = ui.prompt("Tutup Buku", "Silakan masukkan Tahun (contoh: 2026):", ui.ButtonSet.OK_CANCEL);
  if (responsTahun.getSelectedButton() !== ui.Button.OK) return;
  var tahun = responsTahun.getResponseText();
  
  var daftarDawis = ["DAWIS 1", "DAWIS 2", "DAWIS 3"];
  var dataUntukDisimpan = [];
  
  // Mengambil data dari setiap tabel matriks Dawis
  for (var d = 0; d < daftarDawis.length; d++) {
    var sheetDawis = ss.getSheetByName(daftarDawis[d]);
    
    if (sheetDawis) {
      var data = sheetDawis.getDataRange().getValues();
      var jumlahBaris = sheetDawis.getLastRow();
      
      // Mengambil data mulai dari baris ke-5 (karena baris 1-4 adalah header/judul)
      for (var i = 4; i < jumlahBaris; i++) {
        var namaWarga = data[i][1]; // Membaca Kolom B (NAMA)
        var totalSetoran = data[i][33]; // Membaca Kolom AH (Total)
        
        // Jika warga memiliki total setoran lebih dari 0, simpan ke memori sementara
        if (namaWarga && totalSetoran > 0) {
          dataUntukDisimpan.push([new Date(), bulan, tahun, daftarDawis[d], namaWarga, totalSetoran]);
        }
      }
      
      // Proses Mengosongkan Tabel: Menghapus angka dari Tanggal 1 hingga 31
      if (jumlahBaris > 4) {
        // Menghapus data dari baris ke-5, kolom ke-3 (C), sebanyak baris warga, mencakup 31 kolom tanggal
        sheetDawis.getRange(5, 3, jumlahBaris - 4, 31).clearContent();
      }
    }
  }
  
  // Menyimpan semua memori ke tab DATABASE_SEJARAH
  if (dataUntukDisimpan.length > 0) {
    var barisAkhir = sheetSejarah.getLastRow() + 1;
    sheetSejarah.getRange(barisAkhir, 1, dataUntukDisimpan.length, dataUntukDisimpan[0].length).setValues(dataUntukDisimpan);
    ui.alert("Berhasil!", "Rekap " + bulan + " " + tahun + " telah dipindahkan ke DATABASE_SEJARAH dan tabel Dawis telah dikosongkan untuk bulan baru.", ui.ButtonSet.OK);
  } else {
    ui.alert("Perhatian", "Tidak ada data setoran yang ditemukan untuk dipindahkan.", ui.ButtonSet.OK);
  }
}

function getLaporanTahunan(tahunReq) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("DATABASE_SEJARAH");
    
    if (!sheet) {
      return ContentService.createTextOutput(JSON.stringify({status: "error", message: "Tab DATABASE_SEJARAH belum ada."})).setMimeType(ContentService.MimeType.JSON);
    }
    
    var data = sheet.getDataRange().getValues();
    var hasil = [];
    var totalGlobalTahunan = 0;
    
    // Mulai dari baris ke-2 (indeks 1) karena baris 1 adalah tajuk tabel
    // Asumsi Kolom: 0=Waktu, 1=Bulan, 2=Tahun, 3=Dawis, 4=Nama, 5=Nominal
    for (var i = 1; i < data.length; i++) {
      var tahunDiSheet = String(data[i][2]).trim();
      
      if (tahunDiSheet === String(tahunReq).trim()) {
        var nominal = parseFloat(data[i][5]) || 0;
        totalGlobalTahunan += nominal;
        
        hasil.push({
          bulan: String(data[i][1]).trim(),
          dawis: String(data[i][3]).trim(),
          nama: String(data[i][4]).trim(),
          nominal: nominal
        });
      }
    }
    
    return ContentService.createTextOutput(JSON.stringify({
      status: "success",
      total_tahunan: totalGlobalTahunan,
      data: hasil
    })).setMimeType(ContentService.MimeType.JSON);
    
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({status: "error", message: error.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}
